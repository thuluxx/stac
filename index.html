      C program — the stack itself

      the executing line lights up live 
      #include <stdio.h>
#include <stdlib.h>
 
int *stack;      /* the elements           */
int top  = -1;   /* index of the TOP       */
int size;        /* capacity — you decide  */
 
void createStack(int n)
{
    size  = 8;                    /* set from the page */
    stack = (int *)malloc(size * sizeof(int));
    top   = -1;
}
 
int isEmpty(void) { return top == -1; }
int isFull(void)  { return top == size - 1; }
 
void push(int value)
{
    if (isFull()) {                  /* O(1) check */
        printf("Stack overflow\n");
        return;
    }
    stack[++top] = value;            /* O(1) — no shifting */
}
 
int pop(void)
{
    if (isEmpty()) {
        printf("Stack underflow\n");
        return -1;
    }
    return stack[top--];             /* O(1) — only the top */
}
 
int peek(void)
{
    if (isEmpty()) {
        printf("Stack underflow\n");
        return -1;
    }
    return stack[top];               /* O(1) — one known spot */
}

      LIFO — the element pushed last is popped first. Push, Pop and Peek are all O(1): they only ever touch the top index. Overflow = push on a full stack · Underflow = pop / peek on an empty one.
